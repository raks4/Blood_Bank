  name=$(basename "$source" .c); \
  if [ "$name" = main ]; then extra=-Dmain=bloodbank_app_main; else extra=; fi; \
  gcc -std=c99 -D_POSIX_C_SOURCE=200809L -Iinclude -Wall -Wextra -Wpedantic -Wconversion -Wshadow -O0 -g  $extra -c "$source" -o build/tests/$name.o || exit 1; \
done
